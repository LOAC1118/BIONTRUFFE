RÈGLES COMPLÈTES (Copier-coller intégral)
════════════════════════════════════════════════════════════════════════════

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // ─────────────────────────────────────
    // COMPTES ET RÔLES
    // ─────────────────────────────────────
    match /accounts_biontruffle/{email} {
      // Lire son propre compte
      allow read: if request.auth != null && request.auth.email == email;
      
      // Admin peut lire/créer/modifier/supprimer
      allow read, create, update, delete: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.role == 'admin';
    }
    
    // ─────────────────────────────────────
    // BON DE COMMANDE (BDC)
    // ─────────────────────────────────────
    match /bdc_biontruffle/{docId} {
      // Admin : accès complet
      allow read, create, update, delete: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.role == 'admin';
      
      // Commercial : lecture seule
      allow read: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.permissions.hasAny(['bdc_view']);
    }
    
    // ─────────────────────────────────────
    // ENTRÉE EN STOCK
    // ─────────────────────────────────────
    match /stock_entries_biontruffle/{docId} {
      // Admin : accès complet
      allow read, create, update, delete: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.role == 'admin';
      
      // Commercial : lecture seule
      allow read: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.permissions.hasAny(['stock_view']);
    }
    
    // ─────────────────────────────────────
    // CONFIG & STOCK MAP
    // ─────────────────────────────────────
    match /config_biontruffle/{docId} {
      // Admin : accès complet
      allow read, create, update, delete: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.role == 'admin';
      
      // Commercial : lecture seule
      allow read: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email));
    }
    
    // ─────────────────────────────────────
    // BASE CLIENTS
    // ─────────────────────────────────────
    match /clients_biontruffle/{email} {
      // Admin : accès complet
      allow read, create, update, delete: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.role == 'admin';
      
      // Commercial : lecture seule
      allow read: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.permissions.hasAny(['clients_view']);
    }
    
    // ─────────────────────────────────────
    // FACTURES
    // ─────────────────────────────────────
    match /factures_biontruffle/{docId} {
      // Admin : accès complet
      allow read, create, update, delete: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.role == 'admin';
      
      // Commercial : lecture seule
      allow read: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.permissions.hasAny(['invoices_view']);
    }
    
    // ─────────────────────────────────────
    // BONS DE LIVRAISON
    // ─────────────────────────────────────
    match /bls_biontruffle/{docId} {
      // Admin : accès complet
      allow read, create, update, delete: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.role == 'admin';
      
      // Commercial : lecture seule
      allow read: if request.auth != null && 
        exists(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)) &&
        get(/databases/$(database)/documents/accounts_biontruffle/$(request.auth.email)).data.permissions.hasAny(['delivery_view']);
    }
    
    // ─────────────────────────────────────
    // FALLBACK
    // ─────────────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}


EXPLICATIONS DES RÈGLES
════════════════════════════════════════════════════════════════════════════

1. Comptes (accounts_biontruffle)
   ├─ Chacun peut lire son propre compte
   └─ Admin peut tout faire

2. BDC (bdc_biontruffle)
   ├─ Admin: CRUD complet
   └─ Commercial: Lecture seule (permission 'bdc_view')

3. Entrée Stock (stock_entries_biontruffle)
   ├─ Admin: CRUD complet
   └─ Commercial: Lecture seule (permission 'stock_view')

4. Config (config_biontruffle)
   ├─ Admin: CRUD complet
   └─ Commercial: Lecture seule

5. Base Clients (clients_biontruffle)
   ├─ Admin: CRUD complet
   └─ Commercial: Lecture seule (permission 'clients_view')

6. Factures (factures_biontruffle)
   ├─ Admin: CRUD complet
   └─ Commercial: Lecture seule (permission 'invoices_view')

7. Bons de Livraison (bls_biontruffle)
   ├─ Admin: CRUD complet
   └─ Commercial: Lecture seule (permission 'delivery_view')

DÉPANNAGE
════════════════════════════════════════════════════════════════════════════

❌ "Permission denied"
→ Vérifier que l'utilisateur a un compte dans accounts_biontruffle
→ Vérifier le rôle et les permissions

❌ "Missing or insufficient permissions"
→ Vérifier que le compte existe ET que le rôle est correct
→ Vérifier que les règles sont bien sauvegardées

✅ Texte gris "Published" dans l'onglet Règles = OK
